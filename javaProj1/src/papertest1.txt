class Power{
	private int kick;
	private int punch

	public void setKick(int kick){
		this.kick = kick;
	}

	public int getKick(){
		return kick;
	}

	public void setPunch(int punch){
		this.punch = punch;
	}

	public int getPunch(){
		return punch;
	}
}

public class Example{
	public static void main(String[] args){
		Power robot = new Power();
		robot.setKick(10);
		robot.setPunch(20);
	}
}
