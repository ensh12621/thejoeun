public class B extends OddDector{
	public B(int n){
		super(n);
	}

	@Override
	public boolean isOdd(){
		return super.n % 2 == 0 ? false : true;
	}

	public static void main(String[] args){
		B b = new B(10);
		System.out.println(b.isOdd());
	}
}
